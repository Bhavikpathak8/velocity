# VELOCITY Product Management API (Express.js)

This API provides full CRUD (Create, Read, Update, Delete) functionality for products, integrating with the PostgreSQL schema and authentication middleware.

## 1. Routes Overview

| Method | Endpoint | Description | Auth Required |
|---|---|---|---|
| GET | `/api/products` | Fetch all active products | No |
| GET | `/api/products/:slug` | Fetch single product by slug | No |
| POST | `/api/admin/products` | Create new product | Yes (Admin) |
| PUT | `/api/admin/products/:id` | Update existing product | Yes (Admin) |
| DELETE | `/api/admin/products/:id` | Soft delete product | Yes (Admin) |

## 2. Controllers

### A. Create Product (`POST /api/admin/products`)
```javascript
const pool = require('./db');

async function createProduct(req, res) {
  const { name, category_id, description, price, sale_price, sku, stock_quantity, metadata } = req.body;
  const slug = name.toLowerCase().replace(/ /g, '-');

  try {
    const result = await pool.query(
      `INSERT INTO products (name, slug, category_id, description, price, sale_price, sku, stock_quantity, metadata) 
       VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9) RETURNING *`,
      [name, slug, category_id, description, price, sale_price, sku, stock_quantity, JSON.stringify(metadata)]
    );
    res.status(201).json(result.rows[0]);
  } catch (err) {
    res.status(500).json({ error: 'Failed to create product' });
  }
}
```

### B. Update Product (`PUT /api/admin/products/:id`)
```javascript
async function updateProduct(req, res) {
  const { id } = req.params;
  const updates = req.body;
  
  // Dynamic query builder logic here
  try {
    const result = await pool.query(
      `UPDATE products SET name=$1, description=$2, price=$3, stock_quantity=$4, updated_at=CURRENT_TIMESTAMP 
       WHERE id=$5 RETURNING *`,
      [updates.name, updates.description, updates.price, updates.stock_quantity, id]
    );
    res.json(result.rows[0]);
  } catch (err) {
    res.status(500).json({ error: 'Update failed' });
  }
}
```

### C. Delete Product (`DELETE /api/admin/products/:id`)
```javascript
async function deleteProduct(req, res) {
  const { id } = req.params;
  try {
    await pool.query('UPDATE products SET is_active = false WHERE id = $1', [id]);
    res.json({ message: 'Product deactivated successfully' });
  } catch (err) {
    res.status(500).json({ error: 'Deletion failed' });
  }
}
```
